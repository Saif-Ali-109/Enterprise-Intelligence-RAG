# Enterprise Knowledge Intelligence RAG — backend runtime dependencies.
#
# Pinned per Constitution Principle VII. Every entry below buys a capability the
# standard library cannot provide; research.md records the rationale per choice.
#
# Deliberately ABSENT (see research.md and plan.md "Complexity Tracking"):
#   beautifulsoup4, selectolax  — trafilatura already returns lxml, so lxml means
#                                 zero re-parsing. bs4 measured ~14x slower (R-010).
#   langchain, llama-index      — framework-by-default composition is prohibited
#                                 (Constitution VIII); a reviewer must be able to
#                                 read the real chunking and retrieval logic.
#   tiktoken / any tokenizer    — the embedder is hosted and its tokenizer is not
#                                 exposed, so any local count is an approximation
#                                 either way. tokens.py is calibrated against a
#                                 reference tokenizer in tests instead (R-013).
#   redis / any cache           — no measured need at 50-100 pages.

# --- web framework -------------------------------------------------------------
fastapi==0.141.1
uvicorn[standard]==0.40.0

# --- validation and settings ---------------------------------------------------
pydantic==2.13.4
pydantic-settings==2.13.0

# The gold set is validated against contracts/evaluation-dataset.schema.json at
# run time, not only by the test suite: an invalid dataset must refuse the run
# before the first question is asked (T140). That import lives in application
# code, so the library is a runtime dependency — it was in requirements-dev.txt
# until a container build failed with `ModuleNotFoundError: No module named
# 'jsonschema'`, which is the whole argument for building the image in CI rather
# than trusting the virtualenv.
jsonschema==4.26.0

# --- outbound HTTP -------------------------------------------------------------

# --- content extraction --------------------------------------------------------
# lxml is the only parser in the extraction path. trafilatura hands back lxml
# elements, so there is no conversion step and no third parser (R-010).
# trafilatura 2.2.0 requires lxml>=6.1.1; the two are pinned together deliberately
# so a resolver cannot quietly pick an lxml the extractor does not support.
lxml==6.1.3
trafilatura==2.2.0
